cmake_minimum_required(VERSION 3.24)
project(IsayTextService LANGUAGES CXX RC)
option(ISAY_TSF_REF_TRACE "输出组合回调引用跟踪" OFF)
option(ISAY_TSF_TEST_DELAY_PROXY "仅在独立测试构建中延迟回调代理准备" OFF)
add_library(IsayTextService SHARED TextService.cpp PipeEndpoint.cpp CompositionSinkProxy.cpp KeyboardLearningObserver.cpp Version.rc Exports.def)
target_compile_features(IsayTextService PRIVATE cxx_std_20)
target_compile_definitions(IsayTextService PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(ISAY_TSF_REF_TRACE)
  target_compile_definitions(IsayTextService PRIVATE ISAY_TSF_REF_TRACE)
endif()
if(ISAY_TSF_TEST_DELAY_PROXY)
  target_compile_definitions(IsayTextService PRIVATE ISAY_TSF_TEST_DELAY_PROXY)
endif()
target_compile_options(IsayTextService PRIVATE /W4 /utf-8 /permissive- /EHsc)
target_link_libraries(IsayTextService PRIVATE ole32 oleaut32 uuid advapi32 user32)
set_target_properties(IsayTextService PROPERTIES OUTPUT_NAME "iSay.TextService")
